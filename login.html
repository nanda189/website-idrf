const express = require('express'); const sqlite3 = require('sqlite3').verbose(); const bcrypt = require('bcrypt'); const session = require('express-session'); const helmet = require('helmet'); const path = require('path'); const app = express(); const PORT = process.env.PORT || 3000; const SALT_ROUNDS = 12; app.use(helmet()); app.use(express.urlencoded({ extended: false })); app.use(express.json()); app.use(session({ secret: process.env.SESSION_SECRET || 'change_this_secret', resave: false, saveUninitialized: false, cookie: { httpOnly: true, // secure: true, // enable when using HTTPS maxAge: 1000 * 60 * 60 * 2 // 2 hours } })); const db = new sqlite3.Database(path.join(__dirname, 'users.db'), (err) => { if (err) return console.error('DB open error', err); console.log('Connected to SQLite database.'); }); db.run(` CREATE TABLE IF NOT EXISTS users ( id INTEGER PRIMARY KEY AUTOINCREMENT, username TEXT UNIQUE NOT NULL, email TEXT UNIQUE NOT NULL, password_hash TEXT NOT NULL, created_at DATETIME DEFAULT (datetime('now','localtime')) ) `); app.use((req, res, next) => { res.locals.user = req.session.user || null; next(); }); function layout(title, body) { return ` ${title}

${title}

${resLink('/')} ${resLink('/register','Register')} ${resLink('/login','Login')} ${resLink('/profile','Profile')}
${body}
`; } function resLink(url, text) { return `${text || url}`; } app.get('/', (req, res) => { if (req.session.user) { res.send(layout('Home', `

Welcome back, ${escapeHtml(req.session.user.username)}!

Go to profile or Logout

`)); } else { res.send(layout('Home', `

You are not logged in.

Login or Register

`)); } }); app.get('/register', (req, res) => { res.send(layout('Register', `
`)); }); app.post('/register', async (req, res) => { try { const username = (req.body.username || '').trim(); const email = (req.body.email || '').trim().toLowerCase(); const password = req.body.password || ''; if (!username || !email || !password) { return res.send(layout('Register', `

All fields are required.

`)); } if (username.length < 3 || password.length < 6) { return res.send(layout('Register', `

Username or password too short.

`)); } const password_hash = await bcrypt.hash(password, SALT_ROUNDS); const stmt = db.prepare('INSERT INTO users (username, email, password_hash) VALUES (?,?,?)'); stmt.run(username, email, password_hash, function(err) { if (err) { if (err.message && err.message.includes('UNIQUE')) { return res.send(layout('Register', `

Username or email already in use.

`)); } console.error('DB insert error', err); return res.send(layout('Register', `

Server error.

`)); } // auto-login after register req.session.user = { id: this.lastID, username, email }; res.send(layout('Registered', `

Registration successful. Welcome, ${escapeHtml(username)}!

Home | Profile

`)); }); stmt.finalize(); } catch (e) { console.error(e); res.send(layout('Register', `

Server error.

`)); } }); app.get('/login', (req, res) => { res.send(layout('Login', `
`)); }); app.post('/login', (req, res) => { const identifier = (req.body.identifier || '').trim(); const password = req.body.password || ''; if (!identifier || !password) return res.send(layout('Login', `

Missing credentials.

`)); // find by username or email db.get('SELECT id, username, email, password_hash FROM users WHERE username = ? OR email = ?', [identifier, identifier.toLowerCase()], async (err, row) => { if (err) { console.error(err); return res.send(layout('Login', `

Server error.

`)); } if (!row) return res.send(layout('Login', `

Invalid username/email or password.

`)); const match = await bcrypt.compare(password, row.password_hash); if (!match) return res.send(layout('Login', `

Invalid username/email or password.

`)); // success req.session.user = { id: row.id, username: row.username, email: row.email }; res.send(layout('Login success', `

Logged in as ${escapeHtml(row.username)}.

Home | Profile

`)); }); }); app.get('/profile', (req, res) => { if (!req.session.user) return res.redirect('/login'); const u = req.session.user; res.send(layout('Profile', `

${escapeHtml(u.username)}

Email: ${escapeHtml(u.email)}

Logout

`)); }); app.get('/logout', (req, res) => { req.session.destroy((err) => { if (err) console.error('Session destroy error', err); res.clearCookie('connect.sid'); res.redirect('/'); }); }); // Utility: very small HTML escaper function escapeHtml(str) { if (!str) return ''; return String(str) .replace(/&/g, '&') .replace(//g, '>') .replace(/"/g, '"') .replace(/'/g, '''); } // Start server app.listen(PORT, () => { console.log(`Server listening on http://localhost:${PORT}`); });
